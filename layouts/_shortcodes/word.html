{{- $word := trim .Inner " " -}} {{- $id := printf "def-%s" (urlize $word) -}} {{- /* One anchor name per pair is what ties a card to its own word, and only the template knows the word — hence inline styles rather than a rule. */ -}} {{- $anchor := printf "--%s" $id -}} {{- /* One printf, trimmed both ends: any stray newline here lands as a space in the middle of a sentence, right before the punctuation. */ -}} {{- printf `%s — %s` $id $anchor $word $id $anchor $word (.Get "def") | safeHTML -}}
